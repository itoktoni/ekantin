# E-Kanteen Sekolah Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Membangun Sistem e-Kanteen Sekolah (kartu barcode, topup web+tunai simulasi, POS kasir, fee otomatis, limit harian, notifikasi log+retry, laporan CSV+PDF, penarikan, refund) 100% ikut `AGENTS.md`.

**Architecture:** CRUD standar (`BaseModel` + `ControllerTrait` + `Route::auto` + views tiru `pages/users/`) untuk semua entitas; logika uang terisolasi di `app/Actions/Ekantin/` (transaksi atomik + `lockForUpdate` + idempotency); POS = Blade + Alpine (1 POST); notifikasi = `LogChannel` + job retry.

**Tech Stack:** Laravel 13 (PHP ^8.3), Livewire 4 + Flux, Fortify, `lorisleiva/laravel-actions`, `izniburak/laravel-auto-routes`, `bensampo/laravel-enum` ^6.14, `milon/barcode` ^13.1, `barryvdh/laravel-dompdf`, Pest 5.

## Global Constraints

- Semua kolom DB berprefix modul singular: `{module}_{field}`; PK `{module}_id`; FK `{module}_id_{related}` — satu baris tiap aturan, nilai eksak sesuai plan.
- Relasi Eloquent selalu prefix `has` (`hasUser`, `hasGerai`, `hasItems`) — tanpa kecuali.
- Model pakai `protected $fillable` (bukan atribut `#[Fillable]`); casts di `protected function casts(): array`; wajib `field_name()`, `rules()`, `$filterColumns`, `$sortColumns`.
- Controller pakai `ControllerTrait`, `$this->model = $model::getModel()`, `share()` via `array_merge`, CRUD standar via `Route::auto` — tanpa `Route::get/post` untuk CRUD.
- Tiap model wajib Policy di `app/Policies/{Model}Policy.php extends BasePolicy` (tanpa policy = 403).
- Enum bensampo (`extends Enum` + `EnumTrait`), `getDescription(mixed $value): string` via `match`.
- String PHP pakai double quotes, indent 4 spasi, return type selalu dideklarasikan.
- Ekspor = CSV + PDF (dompdf, sudah terinstal) — tanpa paket baru; gateway asli (Midtrans/Xendit/WA) TIDAK diintegrasikan, topup web + notifikasi = simulasi + log.
- Verifikasi tiap task: `php artisan test --filter=<Nama>` + `composer lint:check`.

<?php

namespace App\Actions\Ekantin;

use App\Concerns\PayloadTrait;
use App\Models\FeeConfig;
use App\Models\Kartu;
use App\Models\Produk;
use App\Models\Transaksi;
use Illuminate\Support\Facades\DB;
use Lorisleiva\Actions\Concerns\AsAction;

class ProcessPurchaseAction
{
    use AsAction, PayloadTrait;

    public function handle(array $input): array
    {
        $fee = FeeConfig::aktif();
        $feeTotal = (int) $fee->fee_kebersihan + (int) $fee->fee_keamanan + (int) $fee->fee_pengelolaan + (int) $fee->fee_sistem;

        try {
            return DB::transaction(function () use ($input, $fee, $feeTotal) {
                $kartu = Kartu::where("kartu_barcode", $input["kartu_barcode"])->lockForUpdate()->first();
                if (! $kartu) {
                    return $this->payload(TOAST_FAILED, "Kartu tidak ditemukan.");
                }
                if ($kartu->kartu_status !== "aktif") {
                    return $this->payload(TOAST_FAILED, "Kartu nonaktif, transaksi ditolak.");
                }
                if (! empty($input["idempotency"])) {
                    $ada = Transaksi::where("transaksi_idempotency", $input["idempotency"])->first();
                    if ($ada) {
                        return $this->payload(TOAST_SUCCESS, $ada);
                    }
                }
                $gerai = \App\Models\Gerai::whereKey($input["gerai_id"])->lockForUpdate()->firstOrFail();
                $total = 0;
                $rows = [];
                foreach ($input["items"] as $it) {
                    $p = Produk::whereKey($it["produk_id"])->where("produk_id_gerai", $gerai->gerai_id)->where("produk_status", "tersedia")->first();
                    if (! $p) {
                        return $this->payload(TOAST_FAILED, "Produk tidak tersedia di gerai ini.");
                    }
                    $sub = (int) $p->produk_harga * (int) $it["qty"];
                    $total += $sub;
                    $rows[] = ["nama" => $p->produk_nama, "harga" => (int) $p->produk_harga, "qty" => (int) $it["qty"], "subtotal" => $sub];
                }
                if ((int) $kartu->kartu_saldo < $total + $feeTotal) {
                    return $this->payload(TOAST_FAILED, "Saldo tidak mencukupi. Kurang Rp" . number_format($total + $feeTotal - (int) $kartu->kartu_saldo, 0, ",", "."));
                }
                if (! empty($kartu->kartu_limit_harian)) {
                    $pakai = (int) Transaksi::where("transaksi_id_kartu", $kartu->kartu_id)
                        ->where("transaksi_jenis", "beli")->where("transaksi_status", "berhasil")
                        ->whereDate("created_at", today())->sum("transaksi_total");
                    if ($pakai + $total > (int) $kartu->kartu_limit_harian) {
                        $sisa = (int) $kartu->kartu_limit_harian - $pakai;
                        return $this->payload(TOAST_FAILED, "Limit harian tercapai. Sisa limit Rp" . number_format(max($sisa, 0), 0, ",", "."));
                    }
                }
                $bersih = $total - ((int) $fee->fee_kebersihan + (int) $fee->fee_keamanan + (int) $fee->fee_pengelolaan + (int) $fee->fee_sistem);
                $saldoAkhir = (int) $kartu->kartu_saldo - $total - $feeTotal;
                $trx = Transaksi::create([
                    "transaksi_jenis" => "beli", "transaksi_status" => "berhasil",
                    "transaksi_id_kartu" => $kartu->kartu_id, "transaksi_id_gerai" => $gerai->gerai_id,
                    "transaksi_total" => $total, "transaksi_fee_kebersihan" => $fee->fee_kebersihan,
                    "transaksi_fee_keamanan" => $fee->fee_keamanan, "transaksi_fee_pengelolaan" => $fee->fee_pengelolaan,
                    "transaksi_fee_sistem" => $fee->fee_sistem, "transaksi_bersih" => max($bersih, 0),
                    "transaksi_saldo_akhir" => $saldoAkhir, "transaksi_limit_snapshot" => $kartu->kartu_limit_harian,
                    "transaksi_idempotency" => $input["idempotency"] ?? ("POS-" . $kartu->kartu_id . "-" . time()),
                ]);
                foreach ($rows as $r) {
                    $trx->hasItems()->create(["item_nama" => $r["nama"], "item_harga" => $r["harga"], "item_qty" => $r["qty"], "item_subtotal" => $r["subtotal"]]);
                }
                $kartu->update(["kartu_saldo" => $saldoAkhir]);
                $gerai->increment("gerai_saldo", max($bersih, 0));
                $trx->hasNotif()->create(["notif_saluran" => "log", "notif_status" => "menunggu", "notif_payload" => ["total" => $total, "fee" => $feeTotal, "saldo_akhir" => $saldoAkhir]]);
                if (class_exists(\App\Jobs\KirimNotifikasiJob::class)) {
                    \App\Jobs\KirimNotifikasiJob::dispatch($trx->transaksi_id);
                }

                return $this->payload(TOAST_SUCCESS, $trx->fresh());
            });
        } catch (\Throwable $th) {
            report($th);

            return $this->payload(TOAST_FAILED, $th->getMessage());
        }
    }
}

<?php

namespace App\Jobs;

use App\Models\NotifikasiLog;
use App\Services\Ekantin\NotificationChannelFactory;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Queue\Queueable;

class KirimNotifikasiJob implements ShouldQueue
{
    use Queueable;

    public int $tries = 3;
    public int $backoff = 30;

    public function __construct(public int $transaksiId) {}

    public function handle(): void
    {
        $log = NotifikasiLog::where("notif_id_transaksi", $this->transaksiId)->first();
        if (! $log || $log->notif_status === "terkirim") {
            return;
        }
        $trx = $log->hasTransaksi()->with(["hasKartu.hasUser", "hasGerai", "hasItems"])->first()->hasTransaksi;
        $nama = $trx->hasKartu?->hasUser?->name ?? "Siswa";
        $gerai = $trx->hasGerai?->gerai_nama ?? "-";
        $item = $trx->hasItems->map(fn ($i) => "{$i->item_nama} x{$i->item_qty}")->join(", ");
        $isi = "Ananda {$nama} jajan di {$gerai}: {$item}. Total Rp" . number_format((int) $trx->transaksi_total, 0, ",", ".") . ". Saldo Rp" . number_format((int) $trx->transaksi_saldo_akhir, 0, ",", ".");
        $log->increment("notif_percobaan");
        $ok = NotificationChannelFactory::create("log")->send($nama, "Notifikasi e-Kanteen", $isi);
        $log->update(["notif_status" => $ok ? "terkirim" : ($log->notif_percobaan >= 3 ? "gagal" : "menunggu")]);
        if (! $ok && $log->notif_percobaan < 3) {
            throw new \RuntimeException("Kirim notifikasi gagal, coba lagi.");
        }
    }
}

`ProdukController`: + `getData()` scoping vendor + upload foto via trait aliasing:

```php
use App\Concerns\ControllerTrait;
use App\Enums\Ekantin\ProdukStatusEnum;
use App\Http\Requests\GeneralRequest;
use App\Models\Gerai;
use App\Models\Produk;
use Illuminate\Validation\ValidationException;

class ProdukController extends Controller
{
    use ControllerTrait {
        postCreate as traitPostCreate;
        postUpdate as traitPostUpdate;
    }

    public function __construct(Produk $model) { $this->model = $model::getModel(); }

    protected function share($data = [])
    {
        $gerai = Gerai::query();
        if (auth()->user()?->role === "vendor") {
            $gerai->where("gerai_id_vendor", auth()->id());
        }

        return array_merge(["model" => $this->model, "status" => ProdukStatusEnum::getOptions(), "gerai" => $gerai->pluck("gerai_nama", "gerai_id")], $data);
    }

    protected function getData()
    {
        $q = $this->model->leftJoinRelationship("hasGerai")->filter()->sort();
        if (auth()->user()?->role === "vendor") {
            $ids = Gerai::where("gerai_id_vendor", auth()->id())->pluck("gerai_id");
            $q->whereIn("produk.produk_id_gerai", $ids);
        }

        return $q;
    }

    public function postCreate(GeneralRequest $request)
    {
        $this->mergeFoto($request, null);

        return $this->traitPostCreate($request);
    }

    public function postUpdate(GeneralRequest $request, $id)
    {
        $this->mergeFoto($request, $this->model->findOrFail($id)->produk_foto);

        return $this->traitPostUpdate($request, $id);
    }

    private function mergeFoto(GeneralRequest $request, ?string $existing): void
    {
        if ($request->hasFile("produk_foto")) {
            try {
                $request->merge(["produk_foto" => uploadFile($request->file("produk_foto"), "produk", ["max_size" => 2048])]);
            } catch (\InvalidArgumentException $e) {
                throw ValidationException::withMessages(["produk_foto" => $e->getMessage()]);
            }
        }
    }
}
```
